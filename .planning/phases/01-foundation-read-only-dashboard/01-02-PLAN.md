```python
class AgentStatus(StrEnum):
    ONLINE = "online"
    IDLE = "idle"
    OFFLINE = "offline"
    DND = "dnd"

class TaskStatus(StrEnum):
    QUEUED = "queued"
    RUNNING = "running"
    INPUT_REQUIRED = "input-required"
    AUTH_REQUIRED = "auth-required"
    COMPLETED = "completed"
    FAILED = "failed"
    CANCELED = "canceled"
    REJECTED = "rejected"

@dataclass
class Agent:
    id: str; name: str; status: AgentStatus; last_seen: str | None
    card: AgentCard | None; is_human: bool

@dataclass
class AgentCard:
    description: str; skills: list[AgentSkill]

@dataclass
class AgentSkill:
    name: str; description: str; tags: list[str]

@dataclass
class Task:
    id: str; requester: str; assignee: str; status: TaskStatus
    created_at: str; updated_at: str
```
